[@元歌智能触发]
{
#IF
CheckOnline <$USERNAME>
NOT ISDUMMY
CheckContainsText <$STR(S$元歌挂机)> GJ:
NOT CheckContainsText <$STR(S$元歌挂机)> CX:
#ACT
GOTO @元歌挂机触发
DELAYGOTO 1000 @元歌智能触发
break


[@元歌挂机触发]
#IF
SMALL N$元歌攻击秒 10
#ACT
INC N$元歌攻击秒 1
SENDTOPCHATBOARDMSG 1 0 250 2 ╔┈┈┈┈┈┈┈┈╗
SENDTOPCHATBOARDMSG 1 0 250 2 ║智能挂机正在执行║<$STR(N$元歌挂机次)>
SENDTOPCHATBOARDMSG 1 0 250 2 ╚┈┈┈┈┈┈┈┈╝

#OR
EQUAL <$STR(S$元歌挂机地图<$STR(N$元歌地图计数)>)> 无记录
EQUAL <$STR(S$元歌挂机地图<$STR(N$元歌地图计数)>)> 
#ACT
MOV N$元歌地图计数 0
INC N$元歌地图计数 1

#IF
SMALL N$元歌挂机次 <$STR(N$元歌挂机总)>
#ACT
INC N$元歌挂机次 1
#ELSEACT
MOV N$元歌挂机次 0
INC N$元歌地图计数 1
; ★ 改动2：地图计数超过3时自动重置为1
#IF
LARGE N$元歌地图计数 3
#ACT
MOV N$元歌地图计数 1
map <$STR(S$元歌挂机地图<$STR(N$元歌地图计数)>)>
DELAYGOTO 500 @元歌重启挂机
SENDTOPCHATBOARDMSG 1 0 250 2 ╔┈┈┈┈┈┈┈┈╗
SENDTOPCHATBOARDMSG 1 0 250 2 ║----自动换图----║<$STR(N$元歌地图计数)>
SENDTOPCHATBOARDMSG 1 0 250 2 ╚┈┈┈┈┈┈┈┈╝
break

#IF
CheckContainsText <$STR(S$元歌挂机)> CXHC:
CheckHpPer < 10
#ACT
GoHome
INC S$元歌挂机 CX:
SENDTOPCHATBOARDMSG 1 0 250 2 ╔┈┈┈┈┈┈┈┈╗
SENDTOPCHATBOARDMSG 1 0 250 2 ║----残血回城----║
SENDTOPCHATBOARDMSG 1 0 250 2 ╚┈┈┈┈┈┈┈┈╝

GOTO @元歌挂机残血
break

;不再地图自动传
#IF
not CHECKMAPNAME S$元歌挂机地图<$STR(N$元歌地图计数)>
#ACT
map <$STR(S$元歌挂机地图<$STR(N$元歌地图计数)>)>
DELAYGOTO 500 @元歌重启挂机
SENDTOPCHATBOARDMSG 1 0 250 2 ╔┈┈┈┈┈┈┈┈╗
SENDTOPCHATBOARDMSG 1 0 250 2 ║----自动巡航----║
SENDTOPCHATBOARDMSG 1 0 250 2 ╚┈┈┈┈┈┈┈┈╝
break

;遇人随机
#IF
CheckContainsText <$STR(S$元歌挂机)> YRSJ:
CheckRangeHumCount <$MAP> <$X> <$Y> 10 > 1
#ACT
map <$MAP>
DELAYGOTO 500 @元歌重启挂机
SENDTOPCHATBOARDMSG 1 0 250 2 ╔┈┈┈┈┈┈┈┈╗
SENDTOPCHATBOARDMSG 1 0 250 2 ║----遇人随机----║
SENDTOPCHATBOARDMSG 1 0 250 2 ╚┈┈┈┈┈┈┈┈╝
break

;10秒没攻击附近没怪
#IF
CHECKMAPNAME S$元歌挂机地图<$STR(N$元歌地图计数)>
CheckContainsText <$STR(S$元歌挂机)> GJSJ:
EQUAL <$STR(N$元歌攻击秒)> 10
CheckRangeMonCount <$MAP> <$X> <$Y> 15 < 1
#ACT
MOV N$元歌攻击秒 0
map <$MAP>
DELAYGOTO 100 @元歌重启挂机
SENDTOPCHATBOARDMSG 1 0 250 2 ╔┈┈┈┈┈┈┈┈╗
SENDTOPCHATBOARDMSG 1 0 250 2 ║----无怪随机----║
SENDTOPCHATBOARDMSG 1 0 250 2 ╚┈┈┈┈┈┈┈┈╝

break

;10秒没攻击附近有怪
#IF
CHECKMAPNAME S$元歌挂机地图
CheckContainsText <$STR(S$元歌挂机)> GJSJ:
EQUAL <$STR(N$元歌攻击秒)> 10
CheckRangeMonCount <$MAP> <$X> <$Y> 15 > 0
#ACT
MOV N$元歌攻击秒 0
DELAYGOTO 100 @元歌重启挂机
SENDTOPCHATBOARDMSG 1 0 250 2 ╔┈┈┈┈┈┈┈┈╗
SENDTOPCHATBOARDMSG 1 0 250 2 ║----有怪挂机----║
SENDTOPCHATBOARDMSG 1 0 250 2 ╚┈┈┈┈┈┈┈┈╝

break

[@元歌重启挂机]
#IF
CheckOnline <$USERNAME>
CHECKMAPNAME S$元歌挂机地图<$STR(N$元歌地图计数)>
CheckContainsText <$STR(S$元歌挂机)> GJ:
#ACT
StartAutoPlayGame
break

[@元歌挂机残血]
#IF
CheckHpPer < 90
#ACT
AddhpPer + 5
AddMpPer + 5
SENDTOPCHATBOARDMSG 1 0 250 2 ╔┈┈┈┈┈┈┈┈╗
SENDTOPCHATBOARDMSG 1 0 250 2 ║--泉水回血中5%--║
SENDTOPCHATBOARDMSG 1 0 250 2 ╚┈┈┈┈┈┈┈┈╝

DELAYGOTO 2000 @元歌挂机残血
BREAK
#ELSEACT
DEC S$元歌挂机 CX:
GOTO @元歌智能触发
BREAK
}
