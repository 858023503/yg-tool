[@GM管理界面]
{
#IF
#ACT
; ===== 目标玩家状态检测 =====
#IF
NOT EQUAL <$STR(S$GM目标玩家)>
CheckOnline <$STR(S$GM目标玩家)>
#ACT
MOV S$GM目标状态 [在线]
#ELSEACT
MOV S$GM目标状态 [不在线]
#IF
EQUAL <$STR(S$GM目标玩家)>
#ACT
MOV S$GM目标状态 [未设置]
#IF
#ACT
#say
\ \ \ \ \ \
\<--------------------------------------------------------------------------------/FCOLOR=89>
\ \
\    <◇【GM 管理后台】/FCOLOR=89>
\ \ \
\     <目标玩家（杀怪点用）：/SCOLOR=250><$STR(S$GM目标玩家)>  <$STR(S$GM目标状态)/SCOLOR=254>
\     <请输入目标玩家名称：/SCOLOR=253><点击输入/@@InPutString143>
\ \
\<--------------------------------------------------------------------------------/FCOLOR=89>
\ \
\     <◆ 杀怪点管理（可发目标玩家或自己）/FCOLOR=89>
\     <『+10』/@加10> <『+100』/@加100> <『+1000』/@加1000> <『+5000』/@加5000> <『清零』/@清零杀怪点> <自定义/@@InPutString89>
\ \
\     <◆ 货币发放（对自己生效）/FCOLOR=89><每个版本货币命名不同，刷的时候看背包哪个变动就是哪个/FCOLOR=218>
\     元宝：<点击输入/@@InPutInteger40>  游戏点：<点击输入/@@InPutInteger41>  金币：<点击输入/@@InPutInteger42>
\     金刚石：<点击输入/@@InPutInteger43>  灵符：<点击输入/@@InPutInteger44>  声望：<点击输入/@@InPutInteger45>  荣誉：<点击输入/@@InPutInteger46>
\ \
\     <◆ 物品发放（对自己生效）/FCOLOR=89>
\     物品名：<点击输入/@@InPutString50>（当前：<$STR(S$GM物品)>）  数量：<点击输入/@@InPutInteger47>
\ \
\<--------------------------------------------------------------------------------/FCOLOR=89>

;============================ 目标玩家输入 + 在线校验 ============================
[@InPutString143]
#IF
CheckOnline <$STR(S143)>
#ACT
MOV S$GM目标玩家 <$STR(S143)>
SENDMSG 6 目标玩家已设置为：<$STR(S$GM目标玩家)>
GOTO @GM管理界面
#ELSEACT
MESSAGEBOX 玩家「<$STR(S143)>」不在线或不存在，请确认名字！
break

;============================ 杀怪点管理（目标玩家，立即生效） ============================
[@加10]
#IF
CheckOnline <$STR(S$GM目标玩家)>
#ACT
<$STR(S$GM目标玩家)>.INC U<$str(N$元歌杀怪点)> 10
SENDMSG 6 已给<$STR(S$GM目标玩家)>增加10杀怪点！
GOTO @GM管理界面
#ELSEACT
MESSAGEBOX 目标玩家不在线或未设置！
break

[@加100]
#IF
CheckOnline <$STR(S$GM目标玩家)>
#ACT
<$STR(S$GM目标玩家)>.INC U<$str(N$元歌杀怪点)> 100
SENDMSG 6 已给<$STR(S$GM目标玩家)>增加100杀怪点！
GOTO @GM管理界面
#ELSEACT
MESSAGEBOX 目标玩家不在线或未设置！
break

[@加1000]
#IF
CheckOnline <$STR(S$GM目标玩家)>
#ACT
<$STR(S$GM目标玩家)>.INC U<$str(N$元歌杀怪点)> 1000
SENDMSG 6 已给<$STR(S$GM目标玩家)>增加1000杀怪点！
GOTO @GM管理界面
#ELSEACT
MESSAGEBOX 目标玩家不在线或未设置！
break

[@加5000]
#IF
CheckOnline <$STR(S$GM目标玩家)>
#ACT
<$STR(S$GM目标玩家)>.INC U<$str(N$元歌杀怪点)> 5000
SENDMSG 6 已给<$STR(S$GM目标玩家)>增加5000杀怪点！
GOTO @GM管理界面
#ELSEACT
MESSAGEBOX 目标玩家不在线或未设置！
break

[@清零杀怪点]
#IF
CheckOnline <$STR(S$GM目标玩家)>
#ACT
<$STR(S$GM目标玩家)>.MOV U<$str(N$元歌杀怪点)> 0
SENDMSG 6 已将<$STR(S$GM目标玩家)>的杀怪点清零！
GOTO @GM管理界面
#ELSEACT
MESSAGEBOX 目标玩家不在线或未设置！
break

[@InPutString89]
#IF
LARGE S89 0
CheckOnline <$STR(S$GM目标玩家)>
#ACT
MOV I999 <$STR(S89)>
<$STR(S$GM目标玩家)>.INC U<$str(N$元歌杀怪点)> <$STR(I999)>
SENDMSG 6 已给<$STR(S$GM目标玩家)>增加<$STR(I999)>杀怪点！
GOTO @GM管理界面
#ELSEACT
MESSAGEBOX 数量无效或目标玩家不在线！
break

;============================ 货币发放（对自己生效） ============================
; 元宝
[@InPutInteger40]
#IF
LARGE N40 0
#ACT
GAMEGOLD + <$STR(N40)>
SENDMSG 6 给自己发放<$STR(N40)>元宝！
GOTO @GM管理界面
#ELSEACT
MESSAGEBOX 请输入有效的数量！

[@InPutString50]

#ACT

#IF
LARGE N$GM物品检查 0
#ACT

MESSAGEBOX 物品【<$STR(S50)>】不存在，请确认名称！

#IF
LARGE N47 0
NOT EQUAL <$STR(S$GM物品)>
#ACT
GIVE <$STR(S$GM物品)> <$STR(N47)>
SENDMSG 6 给自己发放<$STR(N47)>个【<$STR(S$GM物品)>】！
MOV S$GM物品
GOTO @GM管理界面
#ELSEACT
MESSAGEBOX 数量无效或未设置物品名！

}
