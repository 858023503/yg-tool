#IF
CHECKSLAVECOUNT < 5

[@战士召唤傀儡]

#ACT
GetCustomItemText 16 S$傀儡召唤名字
;sendmsg 6 通过5<$str(S$傀儡召唤名字)>
mov L$傀儡蛋 [<$str(S$傀儡召唤名字)>]
mov s$傀儡蛋召唤 <$str(L$傀儡蛋[0])>
;sendmsg 6 通过初始<$str(S$傀儡蛋召唤)>
#IF
CheckContainsText <$str(L$傀儡蛋[1])> 速度
#ACT
inc s$傀儡蛋召唤 33 
;sendmsg 6 通过3<$str(S$傀儡蛋召唤)>
#IF
CheckContainsText <$str(L$傀儡蛋[1])> 力量
#ACT
inc s$傀儡蛋召唤 22 
;sendmsg 6 通过3<$str(S$傀儡蛋召唤)>
#IF
#ACT
RECALLMOB <$str(S$傀儡蛋召唤)> 0 180
TakeposW 16
;sendmsg 6 通过2<$str(S$傀儡蛋召唤)>

}
